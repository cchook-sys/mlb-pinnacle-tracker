async function fetchGames() { try { const res = await fetch("https://mlb-pinnacle-tracker.onrender.com/games"); const json = await res.json(); document.getElementById("status").innerText = "最後同步: " + (json.system_updated_at || "未知"); const list = document.getElementById("games-list"); list.innerHTML = ""; const games = json.data || []; games.forEach(function(g) { const bookies = g.bookmakers || []; const pin = bookies.find(function(b) { return b.key === 'pinnacle'; }) || bookies[0]; const pinKey = pin ? pin.key : '無數據'; // 創建容器 const item = document.createElement("div"); item.className = "bg-gray-900 p-4 rounded border border-gray-700 mb-2"; // 使用純文字節點，完全避開字串拼接錯誤 const title = document.createElement("div"); title.className = "font-bold text-lg"; title.textContent = g.away_team + " VS " + g.home_team; const source = document.createElement("div"); source.className = "text-xs text-gray-500"; source.textContent = "來源: " + pinKey; item.appendChild(title); item.appendChild(source); list.appendChild(item); }); } catch (e) { document.getElementById("status").innerText = "錯誤: 無法獲取數據"; console.error(e); } }
