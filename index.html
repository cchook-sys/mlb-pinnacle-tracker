async function fetchData() { const btn = document.getElementById('refresh-btn'); const st = document.getElementById('status-text'); btn.disabled = true; st.innerHTML = '正在更新...'; try { const res = await fetch(API + '/games', { cache: 'no-cache' }); const json = await res.json(); // --- 關鍵除錯：讓我們確認資料長什麼樣 --- console.log("後端回傳的完整資料:", json); // 自動適應結構：不管回傳的是 {data: [...]} 還是 [...] const rawList = Array.isArray(json) ? json : (json.data || []); allGames = rawList.map(item => { // 處理嵌套資料，自動檢查 item.data 或 item 本身 const d = item.data || item; return { home: d.home_team || d.home || 'Unknown', away: d.away_team || d.away || 'Unknown', commence_time: d.commence_time || '', // 加入更多可能的路徑，防止出現 0 total: d.total || d.total_points || 0, over_price: d.over_price || d.over || 0, under_price: d.under_price || d.under || 0, ml_home: d.ml_home_price || d.ml_home || 0, ml_away: d.ml_away_price || d.ml_away || 0 }; }); st.innerHTML = '● 已連線'; renderToday(); } catch(e) { console.error("fetchData 錯誤:", e); st.innerHTML = '連線失敗'; } finally { btn.disabled = false; } }
