async function selectGame(i) { const g = allGames[i]; const pin = g.bookmakers.find(b => b.key === 'pinnacle') || g.bookmakers[0]; const h2h = pin.markets.find(m => m.key === 'h2h'); // 1. 補上「主獨贏」數據顯示 const homeTeam = g.home_team; const homeOdds = h2h.outcomes.find(o => o.name === homeTeam)?.price || '--'; document.getElementById('details-text').innerHTML = `
主隊 (${homeTeam}) 獨贏水位: ${homeOdds}
最後回傳: ${new Date().toLocaleTimeString()}
`; // 2. 繪製圖表 (補上每10分鐘的歷史模擬點) // 注意：若要真正顯示 10 分鐘回傳紀錄，後端需儲存歷史點位，目前我們先用數值填充 const ctx = document.getElementById('oddsChart').getContext('2d'); if(myChart) myChart.destroy(); myChart = new Chart(ctx, { type: 'line', data: { labels: ['16:00', '16:10', '16:20', '16:30'], // 模擬 10 分鐘間隔 datasets: [{ label: '主獨贏水位變化', data: [homeOdds - 5, homeOdds - 2, homeOdds, homeOdds + 3], // 模擬數據點 borderColor: '#fbbf24', tension: 0.3 }] }, options: { responsive: true, maintainAspectRatio: false } }); }
