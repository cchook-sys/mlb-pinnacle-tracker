async function fetchGames() { try { const res = await fetch("https://mlb-pinnacle-tracker.onrender.com/games"); const json = await res.json(); document.getElementById("status").innerText = "最後同步: " + (json.system_updated_at || "未知"); const list = document.getElementById("games-list"); list.innerHTML = ""; const games = json.data || []; games.forEach(function(g) { const bookies = g.bookmakers || []; const pin = bookies.find(function(b) { return b.key === 'pinnacle'; }) || bookies[0]; const pinKey = pin ? pin.key : '無數據'; // 使用最原始的字串拼接，避免反引號編碼問題 const item = document.createElement("div"); item.className = "bg-gray-900 p-4 rounded border border-gray-700 mb-2"; item.innerHTML = "
" + g.away_team + " VS " + g.home_team + "
" + "
來源: " + pinKey + "
"; list.appendChild(item); }); } catch (e) { document.getElementById("status").innerText = "錯誤: 無法獲取數據"; console.error(e); } }
