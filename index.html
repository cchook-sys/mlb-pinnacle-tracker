async function fetchData() { const btn = document.getElementById('refresh-btn'); const st = document.getElementById('status-text'); // 強制顯示正在更新 st.innerHTML = '正在刷新數據...'; btn.disabled = true; try { const res = await fetch(API + '/games'); const json = await res.json(); // 【除錯檢查】在 Console 印出後端回傳的原始結構 console.log("後端回傳資料:", json); // 如果 json.data 是陣列，直接賦值 allGames = Array.isArray(json.data) ? json.data.map(item => { const d = item.data; return { id: item.game_id, home: d.home, away: d.away, commence_time: d.commence_time, latest: { total: d.total, over_juice: d.over_price, under_juice: d.under_price, ml_home: d.ml_home, ml_away: d.ml_away }, snapshot_count: 1, // 預設值 history: [{total: d.total}] }; }) : []; // 渲染畫面 renderToday(); // 更新狀態文字 st.innerHTML = '● 已連線'; document.getElementById('games-count').textContent = allGames.length; } catch(e) { console.error("fetchData 錯誤:", e); st.innerHTML = '連線失敗，請檢查 Console'; } finally { btn.disabled = false; } }
