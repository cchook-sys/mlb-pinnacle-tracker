async function fetchData() { const btn = document.getElementById('refresh-btn'); const st = document.getElementById('status-text'); btn.disabled = true; st.innerHTML = '更新中...'; try { const res = await fetch(API + '/games', { cache: 'no-cache' }); const json = await res.json(); // 確保所有欄位都有預設值，避免 TypeError allGames = (json.data || []).map(item => { const d = item.data || {}; return { id: item.game_id || 'N/A', home: d.home || '—', away: d.away || '—', commence_time: d.commence_time || '', latest: { total: d.total || 0, over_juice: d.over_price || 0, under_juice: d.under_price || 0, ml_home: d.ml_home || 0, ml_away: d.ml_away || 0 } }; }); st.innerHTML = '● 已連線'; renderToday(); } catch(e) { console.error("渲染出錯:", e); st.innerHTML = '渲染失敗'; } finally { btn.disabled = false; } } function renderToday() { const tbody = document.getElementById('today-tbody'); if (!tbody) return; // 保護機制 tbody.innerHTML = allGames.length > 0 ? allGames.map(g => ` ${g.commence_time ? g.commence_time.substring(11, 16) : '--:--'} ${g.home}
${g.away} ${g.latest.total} O:${g.latest.over_juice} / U:${g.latest.under_juice} 主:${g.latest.ml_home} / 客:${g.latest.ml_away} `).join('') : '暫無賽事'; }
