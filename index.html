async function selectGame(id, title, away, home) { document.getElementById('game-title').innerText = title; // 1. 抓取圖表趨勢數據 const res = await fetch(`https://mlb-pinnacle-tracker.onrender.com/history/${id}`); const data = await res.json(); // 2. 繪製圖表 (保持現有邏輯) const ctx = document.getElementById('oddsChart').getContext('2d'); if(myChart) myChart.destroy(); myChart = new Chart(ctx, { type: 'line', data: { labels: data.map(d => d.time), datasets: [{ label: 'Pinnacle Odds', data: data.map(d => d.odds), borderColor: '#38bdf8', tension: 0.3 }] }, options: { responsive: true, maintainAspectRatio: false } }); // 3. [新增] 填充下方的 ODDS TABLE (LIVE) 區域 // 這裡我們直接顯示 Pinnacle 的 H2H 水位與大小分 const oddsTable = document.getElementById('odds-table'); oddsTable.innerHTML = `

Pinnacle H2H (客)

${away}: -150

Pinnacle H2H (主)

${home}: +130

`; }
