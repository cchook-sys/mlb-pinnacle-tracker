// 優化後的選取函數，加入防呆與效能鎖定 async function selectGame(id, title, away, home) { document.getElementById('game-title').innerText = title; try { // 1. 執行請求 const res = await fetch(`https://mlb-pinnacle-tracker.onrender.com/history/${id}`); const data = await res.json(); // 2. 更新表格 (分離處理，避免影響繪圖) const oddsTable = document.getElementById('odds-table'); oddsTable.innerHTML = `

Pinnacle H2H (客)

${away}: -150

Pinnacle H2H (主)

${home}: +130

`; // 3. 繪圖效能處理 const canvas = document.getElementById('oddsChart'); const ctx = canvas.getContext('2d'); if (myChart) { myChart.data.labels = data.map(d => d.time); myChart.data.datasets[0].data = data.map(d => d.odds); myChart.update('none'); // 使用 'none' 模式大幅降低繪圖延遲 } else { myChart = new Chart(ctx, { type: 'line', data: { labels: data.map(d => d.time), datasets: [{ label: 'Pinnacle Odds', data: data.map(d => d.odds), borderColor: '#38bdf8', tension: 0.1 }] }, options: { responsive: true, maintainAspectRatio: false, animation: false } // 關閉動畫，徹底解決卡頓 }); } } catch (err) { console.error("渲染失敗:", err); } }
